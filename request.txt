#!serial
# Inventory only, exact pinned runner AST. No app, tests, build, browser or production traffic.
inventory: set -eu; out="$RUNNER_TEMP/rc-out/full/inventory"; mkdir -p "$out"; test "$(git rev-parse HEAD^)" = 43f611ff12d30901f089e234001bb3b54649b0ef; test "$(git rev-parse HEAD^:)" = 630c84ba75b59ff31efee25a36a5cc61c0e356f9; git rev-parse HEAD HEAD^ HEAD^: > "$out/identity.txt"; git ls-tree -r --full-tree HEAD^ > "$out/product-tree.txt"; test "$(wc -l < "$out/product-tree.txt")" -eq 5502; git diff --name-only HEAD^ HEAD | sort > "$out/qa-diff.txt"; printf '%s\n' .github/workflows/lane-remote-check.yml .rc/request.txt .rc/run.sh | sort > "$out/expected-qa-diff.txt"; diff -u "$out/expected-qa-diff.txt" "$out/qa-diff.txt"; test "$(git hash-object scripts/runAllSims.mjs)" = a7d56d224a472b8eacfe1c4fd2f672f962259d28; sha256sum scripts/runAllSims.mjs package.json package-lock.json > "$out/input-sha256.txt"; start=$(date -u +%FT%TZ); set +e; env -u ONLY -u BROWSER SIM_NETWORK=offline timeout 120 node scripts/runAllSims.mjs --plan-json > "$out/plan.json" 2> "$out/stderr.log"; raw=$?; set -e; end=$(date -u +%FT%TZ); jq -n --arg start "$start" --arg end "$end" --argjson status "$raw" '{start:$start,end:$end,status:$status,scope:"inventory-only",appRuntime:"UNRUN"}' > "$out/process.json"; sha256sum "$out/plan.json" "$out/stderr.log" > "$out/raw-sha256.txt"; gzip -k "$out/plan.json" "$out/stderr.log"; test "$raw" -eq 0; jq -e '((.nodeGroup|type)=="array") and ((.browserGroup|type)=="array") and ((.nodeGroup+.browserGroup|length)==1022) and ((.nodeGroup+.browserGroup|unique|length)==1022)' "$out/plan.json" > "$out/schema-check.txt"; jq -r '.nodeGroup[],.browserGroup[]' "$out/plan.json" | sort > "$out/actual-names.txt"; git ls-tree --name-only HEAD^ scripts/ | sed 's#^scripts/##' | grep -E '^(sim|play|sweep)[A-Za-z0-9]*\.mjs$' | sort > "$out/expected-names.txt"; diff -u "$out/expected-names.txt" "$out/actual-names.txt"; git diff --exit-code HEAD -- > "$out/final-source-diff.txt"; sha256sum "$out/plan.json.gz" "$out/stderr.log.gz" "$out/product-tree.txt" "$out/identity.txt" > "$out/retained-sha256.txt"; jq '{node:(.nodeGroup|length),browser:(.browserGroup|length),total:(.nodeGroup+.browserGroup|length)}' "$out/plan.json"
